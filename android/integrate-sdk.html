集成 SDK ​

本文介绍如何集成 ScribbleForge Android SDK。

开发环境要求 ​

  • Android Studio 4.2 或更高版本
  • Android SDK API 等级 21 或以上
  • Android 5.0 或以上版本的设备

集成步骤 ​

1. 添加 Maven 仓库 ​

在项目根目录的 settings.gradle 文件中添加 Maven 仓库：

groovy
dependencyResolutionManagement {
    repositoriesMode.set(RepositoriesMode.FAIL_ON_PROJECT_REPOS)
    repositories {
        google()
        mavenCentral()
    }
}

2. 添加依赖 ​

在模块的 build.gradle 文件中添加以下依赖：

groovy
dependencies {
    implementation 'io.github.duty-os.forge:forge-all:1.0.0'
}

3. 添加项目权限 ​

在 AndroidManifest.xml 文件中添加以下权限：

xml
<uses-permission android:name="android.permission.INTERNET" />

4. 防止代码混淆 ​

在 proguard-rules.pro 文件中添加以下混淆规则：

-keep class io.agora.board.forge.**{*;}
-dontwarn io.agora.board.forge.**

初始化 SDK ​

1. 创建房间配置 ​

kotlin
val roomOptions = RoomOptions(
    context = applicationContext,
    roomId = "YOUR_ROOM_ID",           // 房间 ID
    roomToken = "YOUR_ROOM_TOKEN",     // 房间 Token
    userId = "YOUR_USER_ID"            // 用户 ID
).apply {
    nickName("用户昵称")                // 可选：用户昵称
    region(RoomRegion.CN)              // 可选：服务区域
    writable(true)                     // 可选：是否可写，默认 true
    
    // 配置消息通信
    socketProvider(RtmSocketProvider(rtmClient))
}

2. 创建并加入房间 ​

kotlin
val room = Room(roomOptions)

room.joinRoom(object : RoomCallback<Boolean> {
    override fun onSuccess(result: Boolean) {
        // 加入房间成功
        Log.i("Room", "加入房间成功")
    }
    
    override fun onFailure(error: RoomError) {
        // 加入房间失败
        Log.e("Room", "加入房间失败: ${error.message}")
    }
})

启动互动白板 ​

kotlin
// 在加入房间成功后启动白板
val whiteboardOption = WhiteboardOption(
    width = 1920,    // 白板宽度
    height = 1080    // 白板高度
)

room.launchWhiteboard(
    option = whiteboardOption,
    appId = "whiteboard_main",  // 可选：应用 ID
    callback = object : RoomCallback<WhiteboardApplication> {
        override fun onSuccess(result: WhiteboardApplication) {
            // 白板启动成功
            val view = result.getView()
            // 将 view 添加到你的布局中
            containerLayout.addView(view)
        }
        
        override fun onFailure(error: RoomError) {
            // 白板启动失败
            Log.e("Whiteboard", "启动失败: ${error.message}")
        }
    }
)

离开房间 ​

kotlin
room.leaveRoom()

完整示例 ​

kotlin
class WhiteboardActivity : AppCompatActivity() {
    private lateinit var room: Room
    
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContentView(R.layout.activity_whiteboard)
        
        // 1. 配置房间选项
        val roomOptions = RoomOptions(
            context = applicationContext,
            roomId = "YOUR_ROOM_ID",
            roomToken = "YOUR_ROOM_TOKEN",
            userId = "user_123"
        ).apply {
            nickName("Alice")
            socketProvider(RtmSocketProvider(rtmClient))
        }
        
        // 2. 创建房间实例
        room = Room(roomOptions)
        
        // 3. 加入房间
        room.joinRoom(object : RoomCallback<Boolean> {
            override fun onSuccess(result: Boolean) {
                // 4. 启动白板
                launchWhiteboard()
            }
            
            override fun onFailure(error: RoomError) {
                Log.e("Room", "加入房间失败: ${error.message}")
            }
        })
    }
    
    private fun launchWhiteboard() {
        val option = WhiteboardOption(width = 1920, height = 1080)
        
        room.launchWhiteboard(option, "whiteboard_main", object : RoomCallback<WhiteboardApplication> {
            override fun onSuccess(result: WhiteboardApplication) {
                result.getView()?.let { view ->
                    findViewById<FrameLayout>(R.id.whiteboard_container).addView(view)
                }
            }
            
            override fun onFailure(error: RoomError) {
                Log.e("Whiteboard", "启动失败: ${error.message}")
            }
        })
    }
    
    override fun onDestroy() {
        super.onDestroy()
        room.leaveRoom()
    }
}
