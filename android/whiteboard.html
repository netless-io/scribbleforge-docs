互动白板 ​

本文介绍如何使用互动白板的核心功能。更多详细功能请参考对应的专题文档或 白板 API 文档。

快速开始 ​

启动白板 ​

kotlin
// 在加入房间成功后启动白板
val whiteboardOption = WhiteboardOption(
    width = 1920,
    height = 1080
)

room.launchWhiteboard(
    option = whiteboardOption,
    appId = "whiteboard_main",
    callback = object : RoomCallback<WhiteboardApplication> {
        override fun onSuccess(result: WhiteboardApplication) {
            // 白板启动成功
            val view = result.getView()
            containerLayout.addView(view)
        }
        
        override fun onFailure(error: RoomError) {
            Log.e("Whiteboard", "启动失败: ${error.message}")
        }
    }
)

获取白板实例 ​

kotlin
// 方法一：通过回调获取
room.launchWhiteboard(option, appId, object : RoomCallback<WhiteboardApplication> {
    override fun onSuccess(result: WhiteboardApplication) {
        val whiteboardApp = result
    }
    override fun onFailure(error: RoomError) {}
})

// 方法二：通过 appId 获取
val whiteboardApp = room.getApp("whiteboard_main") as? WhiteboardApplication

页面操作 ​

添加和删除页面 ​

kotlin
val navigation = whiteboardApp.indexedNavigation

// 添加新页面
navigation.pushPage(goto = true, object : RoomCallback<Int> {
    override fun onSuccess(result: Int) {
        Log.d("Page", "添加成功，总页数: $result")
    }
    override fun onFailure(error: RoomError) {}
})

// 删除页面
navigation.removePage(index = 1, object : RoomCallback<Int> {
    override fun onSuccess(result: Int) {
        Log.d("Page", "删除成功，剩余页数: $result")
    }
    override fun onFailure(error: RoomError) {}
})

页面跳转 ​

kotlin
// 上一页
navigation.prevPage(callback)

// 下一页
navigation.nextPage(callback)

// 跳转到指定页
navigation.gotoPage(index = 2, callback)

权限控制 ​

设置用户权限 ​

kotlin
// 设置用户为只读
val permission = WhiteboardPermission(WhiteboardPermission.NONE)
whiteboardApp.setPermission("user_123", permission)

// 设置基本编辑权限
val permission = WhiteboardPermission(
    WhiteboardPermission.DRAW or
    WhiteboardPermission.EDIT_SELF or
    WhiteboardPermission.DELETE_SELF
)
whiteboardApp.setPermission("user_123", permission)

// 设置所有权限
whiteboardApp.setPermission("user_123", WhiteboardPermission(WhiteboardPermission.ALL))

检查权限 ​

kotlin
whiteboardApp.getPermission(null, object : RoomCallback<WhiteboardPermission> {
    override fun onSuccess(result: WhiteboardPermission) {
        if (result.has(WhiteboardPermission.DRAW)) {
            // 用户有绘制权限
        }
    }
    override fun onFailure(error: RoomError) {}
})

更多权限控制方法请参考：权限控制

视角模式 ​

切换视角模式 ​

kotlin
// 切换到自由模式
whiteboardApp.setViewModeToFree()

// 跟随其他用户
whiteboardApp.setViewModeToFollow(
    followingUserId = "teacher_id",
    userId = room.userId
)

// 切换到主视图模式
whiteboardApp.setViewModeToMain()

获取当前视角模式 ​

kotlin
whiteboardApp.getViewMode(null, object : RoomCallback<String> {
    override fun onSuccess(result: String) {
        when (result) {
            "free" -> Log.d("ViewMode", "自由模式")
            "main" -> Log.d("ViewMode", "主视图模式")
            else -> Log.d("ViewMode", "跟随用户: $result")
        }
    }
    override fun onFailure(error: RoomError) {}
})

更多视角模式说明请参考：白板视角模式

撤销和重做 ​

执行撤销重做 ​

kotlin

whiteboardApp.undo()

// 重做
whiteboardApp.redo()

监听撤销重做栈 ​

kotlin
whiteboardApp.addListener(object : SimpleWhiteboardListener() {
    override fun onUndoStackLengthUpdate(
        whiteboard: WhiteboardApplication,
        length: Int
    ) {
        Log.d("Undo", "撤销栈长度: $length")
        // 更新撤销按钮状态
        undoButton.isEnabled = length > 0
    }
    
    override fun onRedoStackLengthUpdate(
        whiteboard: WhiteboardApplication,
        length: Int
    ) {
        Log.d("Redo", "重做栈长度: $length")
        // 更新重做按钮状态
        redoButton.isEnabled = length > 0
    }
})

事件监听 ​

添加事件监听器 ​

kotlin
whiteboardApp.addListener(object : SimpleWhiteboardListener() {
    
    // 工具信息变化
    override fun onToolInfoUpdate(
        whiteboard: WhiteboardApplication,
        toolInfo: WhiteboardToolInfo
    ) {
        Log.d("Tool", "工具更新: ${toolInfo.tool}")
    }
    
    // 页面信息变化
    override fun onPageInfoUpdate(
        whiteboard: WhiteboardApplication,
        activePageIndex: Int,
        pageCount: Int
    ) {
        Log.d("Page", "当前页: ${activePageIndex + 1}/$pageCount")
    }
    
    // 权限变化
    override fun onPermissionUpdate(
        whiteboard: WhiteboardApplication,
        userId: String,
        value: WhiteboardPermission
    ) {
        Log.d("Permission", "用户 $userId 权限变化")
    }
    
    // 元素选中
    override fun onElementSelected(
        whiteboard: WhiteboardApplication,
        selection: WhiteboardElementSelection
    ) {
        Log.d("Element", "元素被选中: ${selection.uuid}")
    }
    
    // 元素取消选中
    override fun onElementUnselected(whiteboard: WhiteboardApplication) {
        Log.d("Element", "元素取消选中")
    }
    
    // 撤销栈变化
    override fun onUndoStackLengthUpdate(
        whiteboard: WhiteboardApplication,
        length: Int
    ) {
        Log.d("Undo", "撤销栈: $length")
    }
    
    // 重做栈变化
    override fun onRedoStackLengthUpdate(
        whiteboard: WhiteboardApplication,
        length: Int
    ) {
        Log.d("Redo", "重做栈: $length")
    }
    
    // 白板错误
    override fun onWhiteboardError(
        whiteboard: WhiteboardApplication,
        errorCode: Int,
        errorMessage: String
    ) {
        Log.e("Whiteboard", "错误: $errorCode - $errorMessage")
    }
})

最佳实践 ​

  1. 初始化时机：在 Room 成功加入后再启动白板
  2. 视图管理：及时添加白板视图到布局中
  3. 事件监听：使用 SimpleWhiteboardListener 只实现需要的回调
  4. 权限检查：执行敏感操作前检查用户权限

注意事项 ​

  1. 坐标系统：白板使用自己的坐标系统，与设备屏幕坐标不同
  2. 同步机制：所有操作会自动同步到其他用户
  3. 生命周期：白板会随着 Room 的生命周期自动管理
