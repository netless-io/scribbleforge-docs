Whiteboard 页面管理指南 ​

概述 ​

Netless Forge Whiteboard 提供了完整的页面管理系统，支持多页面的创建、删除、跳转等操作。页面管理分为两个层次：

  1. 基础页面操作：通过白板实例直接进行页面管理，使用字符串 ID 标识页面
  2. 顺序页面管理：通过 IndexedNavigation 提供类似 PowerPoint 的页面索引管理，支持上一页、下一页等操作

每个页面都是独立的画布空间，用户可以在不同页面上绘制不同的内容，支持实时的多用户协作。

安全删除页面 ​

typescript
// 安全删除页面函数（防止删除最后一页）
function safeRemovePage(index: number): boolean {
  const totalPages = whiteboard.indexedNavigation.pageCount;

  if (totalPages <= 1) {
    console.warn('无法删除页面：至少需要保留一页');
    return false;
  }

  if (index < 0 || index >= totalPages) {
    console.warn('无法删除页面：页面索引超出范围');
    return false;
  }

  const currentPageIndex = whiteboard.indexedNavigation.currentPageIndex;
  const newPageCount = whiteboard.indexedNavigation.removePage(index);
  console.log(`成功删除第${index + 1}页，剩余${newPageCount}页`);

  // 如果删除的是当前页，自动跳转到前一页
  if (index === currentPageIndex && index > 0) {
    whiteboard.indexedNavigation.gotoPage(index - 1);
  }

  return true;
}

// 使用示例
safeRemovePage(2); // 删除第3页

页面导航工具类 ​

typescript
class PageNavigator {
  private whiteboard: Whiteboard;

  constructor(whiteboard: Whiteboard) {
    this.whiteboard = whiteboard;
  }

  // 获取页面信息
  get currentPage(): number {
    return this.whiteboard.indexedNavigation.currentPageIndex;
  }

  get totalPages(): number {
    return this.whiteboard.indexedNavigation.pageCount;
  }

  // 跳转方法
  first(): void {
    this.whiteboard.indexedNavigation.gotoPage(0);
  }

  last(): void {
    const lastIndex = this.totalPages - 1;
    this.whiteboard.indexedNavigation.gotoPage(lastIndex);
  }

  next(): boolean {
    if (this.canGoNext()) {
      this.whiteboard.indexedNavigation.nextPage();
      return true;
    }
    return false;
  }

  prev(): boolean {
    if (this.canGoPrev()) {
      this.whiteboard.indexedNavigation.prevPage();
      return true;
    }
    return false;
  }

  goTo(index: number): boolean {
    if (index >= 0 && index < this.totalPages) {
      this.whiteboard.indexedNavigation.gotoPage(index);
      return true;
    }
    return false;
  }

  // 状态检查
  canGoNext(): boolean {
    return this.currentPage < this.totalPages - 1;
  }

  canGoPrev(): boolean {
    return this.currentPage > 0;
  }

  // 页面信息
  getPageInfo(): { current: number; total: number; isFirst: boolean; isLast: boolean } {
    return {
      current: this.currentPage,
      total: this.totalPages,
      isFirst: this.currentPage === 0,
      isLast: this.currentPage === this.totalPages - 1
    };
  }
}

// 使用示例
const navigator = new PageNavigator(whiteboard);

console.log('页面信息:', navigator.getPageInfo());
navigator.goTo(2); // 跳转到第3页
navigator.first(); // 跳转到第一页
navigator.last();  // 跳转到最后一页

  const nextButton = document.getElementById('next-button') as HTMLButtonElement;

    prevButton.disabled = currentIndex === 0;

  if (nextButton) {
    nextButton.disabled = currentIndex === totalPages - 1;
  }
}

function updatePageCounter(pageCount: number): void {
  const counter = document.getElementById('page-counter');
  if (counter) {
    counter.textContent = `共 ${pageCount} 页`;
  }
}

综合事件处理示例 ​

typescript
class PageEventManager {
  private whiteboard: Whiteboard;
  private eventListeners: Array<() => void> = [];

  constructor(whiteboard: Whiteboard) {
    this.whiteboard = whiteboard;
    this.setupEventListeners();
  }

  private setupEventListeners(): void {
    // 监听页面索引变化
    const onActivePageChange = (pageIndex: number) => {
      console.log('页面索引变化:', pageIndex);
      this.handlePageChange(pageIndex);
    };

    // 监听总页数变化
    const onPageCountChange = (pageCount: number) => {
      console.log('总页数变化:', pageCount);
      this.handlePageCountChange(pageCount);
    };

    // 注册监听器
    this.whiteboard.indexedNavigation.on('activePageChange', onActivePageChange);
    this.whiteboard.indexedNavigation.on('pageCountChange', onPageCountChange);

    // 保存监听器引用，便于清理
    this.eventListeners.push(
      () => this.whiteboard.indexedNavigation.off('activePageChange', onActivePageChange),
      () => this.whiteboard.indexedNavigation.off('pageCountChange', onPageCountChange)
    );
  }

  private handlePageChange(pageIndex: number): void {
    // 更新UI
    this.updatePageDisplay(pageIndex);

    // 记录页面访问
    this.logPageAccess(pageIndex);

    // 触发自定义事件
    this.onPageChanged(pageIndex);
  }

  private handlePageCountChange(pageCount: number): void {
    // 更新页面控件
    this.updatePageControls(pageCount);

    // 显示通知
    this.showPageCountNotification(pageCount);
  }

  private updatePageDisplay(pageIndex: number): void {
    const totalPages = this.whiteboard.indexedNavigation.pageCount;
    console.log(`UI更新: 显示第 ${pageIndex + 1}/${totalPages} 页`);
  }

  private updatePageControls(pageCount: number): void {
    console.log(`UI更新: 更新页面控件，总共 ${pageCount} 页`);
  }

  private logPageAccess(pageIndex: number): void {
    console.log(`页面访问日志: 用户访问了第 ${pageIndex + 1} 页`);
  }

  private showPageCountNotification(pageCount: number): void {
    console.log(`页面数变化通知: 当前共有 ${pageCount} 页`);
  }

  protected onPageChanged(pageIndex: number): void {
    // 子类可以重写此方法
  }

  // 清理事件监听器
  public dispose(): void {
    this.eventListeners.forEach(cleanup => cleanup());
    this.eventListeners = [];
  }
}

// 使用示例

1. 权限控制 ​

页面操作需要相应的权限：

// 检查权限
function checkPageOperationPermission(operation: string): boolean {
  const permissions = whiteboard.permissions.getPermissionValue();
  const hasMainView = permissions & WhiteboardPermissionFlag.mainView;

  if (!hasMainView && (operation === 'add' || operation === 'delete')) {
    console.warn(`没有权限执行页面${operation}操作`);
    return false;
  }

  return true;
}

// 安全添加页面
function safeAddPage(): boolean {
  if (!checkPageOperationPermission('add')) {
    return false;
  }

  whiteboard.indexedNavigation.pushPage();
  return true;
}

4. 并发操作处理 ​

typescript
// 避免频繁的页面操作
const pageOperationQueue: Array<() => void> = [];
let isProcessing = false;

function enqueuePageOperation(operation: () => void): void {
  pageOperationQueue.push(operation);
  processQueue();
}

function processQueue(): void {
  if (isProcessing || pageOperationQueue.length === 0) {
    return;
  }

  isProcessing = true;

  const operation = pageOperationQueue.shift()!;
  operation();

  // 延迟处理下一个操作，避免冲突
  setTimeout(() => {
    isProcessing = false;
    processQueue();
  }, 100);
}

// 使用示例
enqueuePageOperation(() => {
  whiteboard.indexedNavigation.pushPage();
});

enqueuePageOperation(() => {
  whiteboard.indexedNavigation.gotoPage(0);
});

function batchPageOperations(operations: Array<() => void>): void {

6. 错误处理和恢复 ​

typescript
// 页面操作错误处理
function safePageOperation<T>(
  operation: () => T,
  fallback?: () => T,
  errorMessage?: string
): T | null {
  try {
    return operation();
  } catch (error) {
    console.error(errorMessage || '页面操作失败:', error);

    if (fallback) {
      try {
        return fallback();
      } catch (fallbackError) {
        console.error('fallback操作也失败:', fallbackError);
      }
    }

    return null;
  }
}

// 使用示例
const result = safePageOperation(
  () => whiteboard.indexedNavigation.pushPage(),
  () => whiteboard.indexedNavigation.pageCount, // fallback: 返回当前页数
  '添加页面失败'
);

if (result !== null) {
  console.log('操作成功，当前页数:', result);
}

// 页面相关资源清理
class PageResourceManager {
  private eventCleanups: Array<() => void> = [];

  constructor(private whiteboard: Whiteboard) {
    this.setupEventListeners();
  }

  private setupEventListeners(): void {
    const onPageChange = (index: number) => {
      console.log('页面变化:', index);
    };

    this.whiteboard.indexedNavigation.on('activePageChange', onPageChange);

    // 保存清理函数
    this.eventCleanups.push(() => {
      this.whiteboard.indexedNavigation.off('activePageChange', onPageChange);
    });
  }

  // 清理所有资源
  public dispose(): void {
    this.eventCleanups.forEach(cleanup => cleanup());
    this.eventCleanups = [];
    console.log('页面资源已清理');
  }
}

// 使用示例
const pageResource = new PageResourceManager(whiteboard);
