Whiteboard 教具使用教程 ​

本文档详细介绍 Netless Forge Whiteboard 中各种教具的使用方法，包括教具切换、属性设置、事件监听等。

教具列表 ​

Whiteboard 提供了以下 13 种教具工具：

工具名称工具类型主要功能描述适用场景
曲线工具curve自由绘制平滑曲线，支持手写和鼠标绘制手写笔记、自由绘画、签名
矩形工具rectangle绘制矩形和正方形，支持填充和边框框架标注、区域划分、流程图
选择工具selector选择和操作白板上的元素，支持移动、缩放、旋转编辑元素、调整位置、改变大小
直线工具line绘制直线，支持设置箭头样式连接线、标记指示、几何图形
箭头工具arrow绘制带箭头的直线，支持设置起点和终点箭头流程图、方向指示、关系标注
文本工具text添加和编辑文本内容，支持字体大小和颜色设置文字说明、标注、笔记记录
椭圆工具ellipse绘制椭圆和圆形，支持填充和边框圆形标注、气泡框、装饰元素
三角形工具triangle绘制三角形，支持填充和边框流程图符号、警示标记、装饰
橡皮擦工具eraser擦除白板上的元素修正错误、清除内容、编辑修改
激光笔工具laser激光指示器，用于临时标注和讲解演讲讲解、重点标注、临时指示
抓手工具grab拖拽和移动画布视图画布导航、视图调整、内容浏览
指针工具pointer鼠标指针工具，用于一般操作基础操作、点击选择、导航

切换教具 ​

设置当前教具 ​

typescript
import { Whiteboard, WhiteboardApplication } from '@netless/forge-whiteboard';

// 启动白板应用
const whiteboard = await room.applicationManager.launchApplication(
  WhiteboardApplication,
  {
    width: 1920,
    height: 1080,
    defaultToolbarStyle: {
      tool: 'selector' // 默认工具
    }
  },
  'MainWhiteboard'
);

// 切换到曲线工具
whiteboard.setCurrentTool('curve');

// 切换到矩形工具
whiteboard.setCurrentTool('rectangle');

// 切换到选择工具
whiteboard.setCurrentTool('selector');

// 切换到橡皮擦
whiteboard.setCurrentTool('eraser');

// 切换到激光笔
whiteboard.setCurrentTool('laser');

// 切换到抓手工具
whiteboard.setCurrentTool('grab');

获取当前教具 ​

typescript
// 获取当前选中的工具
const currentTool = whiteboard.getCurrentTool();
console.log('当前工具:', currentTool); // 输出: 'curve' | 'rectangle' | 'selector' | ...

设置教具属性 ​

线条颜色（描边颜色） ​

typescript
// 设置描边颜色
whiteboard.strokeColor = '#FF5722'; // 橙红色
whiteboard.strokeColor = '#2196F3'; // 蓝色
whiteboard.strokeColor = '#4CAF50'; // 绿色
whiteboard.strokeColor = '#000000'; // 黑色

// 也可以使用 RGB 格式
whiteboard.strokeColor = 'rgb(255, 87, 34)';
whiteboard.strokeColor = 'rgba(33, 150, 243, 0.8)'; // 带透明度

填充颜色 ​

typescript
// 设置填充颜色
whiteboard.fillColor = '#FFC107'; // 黄色
whiteboard.fillColor = '#9C27B0'; // 紫色

// 设置为无填充（透明）
whiteboard.fillColor = null; // 或者 'transparent'

线条粗细 ​

typescript
// 设置线条粗细
whiteboard.strokeWidth = 2;   // 细线
whiteboard.strokeWidth = 4;   // 标准线
whiteboard.strokeWidth = 8;   // 粗线
whiteboard.strokeWidth = 16;  // 很粗的线

// 橡皮擦工具的擦除大小也会受到影响

字体大小 ​

typescript
// 设置字体大小（主要影响文本工具）
whiteboard.fontSize = 16; // 小字体
whiteboard.fontSize = 24; // 标准字体
whiteboard.fontSize = 32; // 大字体
whiteboard.fontSize = 48; // 很大字体

虚线样式 ​

typescript
// 设置虚线样式
whiteboard.dashArray = [];           // 实线
whiteboard.dashArray = [5, 5];       // 虚线（5像素线段，5像素间隔）
whiteboard.dashArray = [10, 5];      // 长虚线
whiteboard.dashArray = [2, 2];       // 点线
whiteboard.dashArray = [15, 10, 5, 10]; // 复杂虚线样式

箭头样式 ​

typescript
// 针对特定元素设置箭头样式
const pageId = whiteboard.currentPageId();
const elementId = 'some-element-id';

// 设置起点箭头
whiteboard.setElementAttribute(pageId, elementId, 'headArrow', 'normal');
whiteboard.setElementAttribute(pageId, elementId, 'headArrow', 'none');

// 设置终点箭头
whiteboard.setElementAttribute(pageId, elementId, 'tailArrow', 'normal');
whiteboard.setElementAttribute(pageId, elementId, 'tailArrow', 'none');

字体设置 ​

typescript
// 设置字体族
whiteboard.fontFamily = 'Arial';
whiteboard.fontFamily = 'Times New Roman';
whiteboard.fontFamily = 'Courier New';
whiteboard.fontFamily = 'Microsoft YaHei'; // 微软雅黑
whiteboard.fontFamily = 'SimSun'; // 宋体

监听教具变化 ​

监听工具栏样式变化 ​

typescript
// 监听工具栏样式变化事件
whiteboard.on('toolbarStyleChange', (style) => {
  console.log('工具栏样式变化:', style);
  console.log('当前工具:', style.tool);
  console.log('当前颜色:', style.strokeColor);
  console.log('线条粗细:', style.strokeWidth);
});

监听元素选择事件 ​

typescript
// 监听元素被选中事件
whiteboard.on('elementSelected', (userId, pageId, elementId, attributes, boundingRect) => {
  console.log('元素被选中:', {
    userId,
    pageId,
    elementId,
    attributes,
    boundingRect
  });

  // 可以根据选中的元素调整工具栏
  if (attributes.includes('headArrow')) {
    // 显示箭头设置选项
  }
});

// 监听元素取消选中事件
whiteboard.on('elementDeselected', (userId) => {
  console.log('元素取消选中:', userId);
});

监听撤销/重做栈变化 ​

typescript
// 监听撤销栈变化
whiteboard.on('undoStackLength', (length) => {
  console.log('撤销栈长度:', length);
  // 可以根据长度启用/禁用撤销按钮
});

// 监听重做栈变化
whiteboard.on('redoStackLength', (length) => {
  console.log('重做栈长度:', length);
  // 可以根据长度启用/禁用重做按钮
});

监听工具特定事件 ​

typescript
// 监听抓手工具事件
whiteboard.on('grabDown', () => {
  console.log('抓手工具按下');
});

whiteboard.on('grabUp', () => {
  console.log('抓手工具释放');
});

监听输入类型变化 ​

typescript
// 监听输入类型变化（手写笔检测）
whiteboard.on('inputTypeChange', (type) => {
  console.log('输入类型变化:', type);

  }
});

完整的教具工具栏示例 ​

typescript
import { WhiteboardApplication, WhiteboardToolType } from '@netless/forge-whiteboard';

class WhiteboardToolbar {
  private whiteboard: Whiteboard;
  private currentColor = '#2196F3';
  private currentStrokeWidth = 4;

  constructor(whiteboard: Whiteboard) {
    this.whiteboard = whiteboard;
    this.setupEventListeners();
  }

  // 工具按钮点击处理
  selectTool(tool: WhiteboardToolType) {
    this.whiteboard.setCurrentTool(tool);
    console.log('切换到工具:', tool);
  }

  // 颜色选择处理
  selectColor(color: string) {
    this.currentColor = color;
    this.whiteboard.strokeColor = color;
    console.log('设置颜色:', color);
  }

  // 线条粗细处理
  selectStrokeWidth(width: number) {
    this.currentStrokeWidth = width;
    this.whiteboard.strokeWidth = width;
    console.log('设置线条粗细:', width);
  }

  // 填充颜色处理
  selectFillColor(color: string | null) {
    this.whiteboard.fillColor = color;
    console.log('设置填充颜色:', color);
  }

  // 虚线样式处理
  selectDashArray(pattern: number[]) {
    this.whiteboard.dashArray = pattern;
    console.log('设置虚线样式:', pattern);
  }

  // 撤销操作
  undo() {
    this.whiteboard.undo();
  }

  // 重做操作
  redo() {
    this.whiteboard.redo();
  }

  // 清空当前页面
  clearCurrentPage() {
    this.whiteboard.clearPage();
  }

  // 设置事件监听
  private setupEventListeners() {
    // 监听工具栏变化以同步UI状态
    this.whiteboard.on('toolbarStyleChange', (style) => {
      console.log('工具栏状态同步:', style);
      // 这里可以更新UI控件的状态
    });

    // 监听撤销/重做状态
    this.whiteboard.on('undoStackLength', (length) => {
      console.log('撤销可用:', length > 0);
    });

    this.whiteboard.on('redoStackLength', (length) => {
      console.log('重做可用:', length > 0);
    });
  }
}

// 使用示例
const toolbar = new WhiteboardToolbar(whiteboard);

// 切换工具
toolbar.selectTool('curve');      // 切换到画笔
toolbar.selectTool('rectangle');  // 切换到矩形
toolbar.selectTool('text');       // 切换到文本
toolbar.selectTool('eraser');     // 切换到橡皮擦

// 设置样式
toolbar.selectColor('#FF5722');           // 设置红色
toolbar.selectStrokeWidth(6);              // 设置粗线
toolbar.selectFillColor('#FFC107');        // 设置黄色填充
toolbar.selectDashArray([5, 5]);          // 设置虚线

性能模式 ​

在低端设备上使用复杂教具（如曲线工具）时，可能会出现性能问题。Whiteboard 提供了性能模式选项，可以在创建白板时启用，以提升绘制性能。

启用性能模式

typescript
// 启动参数
const whiteboard = await room.applicationManager.launchApplication(
  WhiteboardApplication,
  {
    width: 1920,
    height: 1080,
    defaultToolbarStyle: {
      tool: 'selector' // 默认工具
    },
    performance: true
  },
  'MainWhiteboard'
);

// 动态设置

whiteboard.setPerformanceMode(true); // 启用性能模式
whiteboard.setPerformanceMode(false); // 关闭性能模式

whiteboard.isPerformanceMode(); // 获取当前性能模式状态

性能模式行为差异 ​

性能模式启用后, 会降低数据同步频率, 与非性能模式相比存在如下差异

对于开启了性能模式的客户端:

  1. 曲线以及橡皮擦工具产生的点数据优先使用本地数据, 本地数据每 80 个点(或者100ms)同步一次到其他客户端
  2. 矩形, 椭圆, 三角形等形状工具数据优先使用本地数据, 本地数据在 100ms 内没变化的情况下，才同步一次到其他客户端
  3. 激光笔因为实时性要求较高, 不受性能模式影响
  4. 文本工具无影响

对于接收性能模式客户端数据的远端:

  1. 曲线以及橡皮擦工具的点数据会进行插值处理, 以保证曲线的平滑度. 视觉上是间隔一定时间, 开始绘制一段平滑的曲线
  2. 矩形, 椭圆, 三角形等工具的绘制过程会被简化为仅同步最终结果, 视觉上是直接显示最终图形
  3. 文本工具无影响

性能提升 ​

性能模式启用后, 以 mac m4 pro 设备测试, 在高压力场景下, CPU 使用率可降低约 45%-50%
